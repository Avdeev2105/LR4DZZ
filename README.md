Алгоритм решения:
Начало.
Ввести номер рабочего места Анны — A.
Ввести номер рабочего места Бориса — B.
Вычислить чётность: ann = (A mod 2 = 0), bor = (B mod 2 = 0).
Проверить условие: чётности различаются?
Если да, включить кофе-машину, вывести «Кофе готовится».
Если нет, не включать, вывести «Кофе не готовится».
Конец.

Код на языке с: 

#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>

#include<locale.h>

int main(void) {

    setlocale(LC_ALL, "RUS");
    
    int a, b;
    
    printf("Введите номер рабочего места Анны (A): ");
    
    if (scanf("%d", &a) != 1) {
    
        printf("Ошибка: введено не число.\n");
        
        return 1;
        
    }
    
    printf("Введите номер рабочего места Бориса (B): ");
    
    if (scanf("%d", &b) != 1) {
    
        printf("Ошибка: введено не число.\n");
        
        return 1;
        
    }
    
    if ((a % 2) != (b % 2)) {
    
        printf("Ровно один из программистов чётный — кофе готовится!\n");
        
    }
    
    else {
    
        printf("Оба чётные или оба нечётные — кофе не готовится.\n");
        
    }

    return 0;
}
